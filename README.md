```python
import os
import sys
from flask import Flask

app = Flask(__name__)

port = int(
    sys.argv[1].split("=")[1]
    if len(sys.argv) > 1
    else os.getenv("PORT", 5000)
)

@app.route("/")
def home():
    return "Hello from APIH9S5!"

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=port)
```
